<script>
  // ⚠️ 7UT LIEN DYAL WEB APP DYALK HNA:
  const WEB_APP_URL = "https://script.google.com/macros/s/AKfycbzWuVErdcuFiW31TIMwWOfr5nwZwk-rQRu2awAI3056fDVY3PJMJmw85tCa5C9-k0cA/exec"; // Ila bdltoh direct f code l-foq

  function formatDateToDDMMAAAA(dateString) {
    if (!dateString) return "";
    var parts = dateString.split("-"); // YYYY-MM-DD
    return parts[2] + "/" + parts[1] + "/" + parts[0]; // JJ/MM/AAAA
  }

  function startVerification() {
    var num = document.getElementById('numAttestation').value;
    var rawDate = document.getElementById('dateSession').value;
    var captcha = document.getElementById('captchaInput').value;
    
    var errBox = document.getElementById('errorContainer');
    var resBox = document.getElementById('resultsContainer');
    
    errBox.style.display = "none";
    resBox.style.display = "none";
    
    if(!num || !rawDate || !captcha) {
      alert("Veuillez remplir tous les champs !");
      return;
    }

    var date = formatDateToDDMMAAAA(rawDate);
    
    errBox.style.color = "#3170af";
    errBox.style.borderColor = "#3170af";
    errBox.style.background = "#e6f0fa";
    errBox.innerText = "Vérification en cours...";
    errBox.style.display = "block";
    
    fetch(WEB_APP_URL, {
      method: "POST",
      mode: "no-cors",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({ numAttestation: num, dateSession: date, captchaInput: captcha })
    })
    .then(function() {
      errBox.style.display = "none";
      resBox.innerHTML = "<h3 style='color:#28a745; margin:0;'>Demande envoyée !</h3>" +
                         "<p style='font-size:14px; color:#555;'>Cliquez ci-dessous pour voir le résultat final :</p>" +
                         "<a href='https://github.com' target='_blank' class='btn-redirect'>📄 Voir le résultat final</a>";
      resBox.style.display = "block";
    })
    .catch(function(e) {
      errBox.style.color = "#dc3545";
      errBox.style.borderColor = "#dc3545";
      errBox.style.background = "#fdf3f4";
      errBox.innerText = "Erreur de connexion.";
    });
  }
</script>
